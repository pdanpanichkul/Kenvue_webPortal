                <th class="px-4 py-3">Supplier Code</th>
                <th class="px-4 py-3">Supplier Name</th>
                <th class="px-4 py-3 text-right">Action</th>
              </tr>
            </thead>
            <tbody id="vendor-master-tbody" class="divide-y divide-slate-100">

            </tbody>
          </table>

  <!-- ==========================================
       JAVASCRIPT LOGIC
       ========================================== -->
  <script>
    // [1] Icon Initialization
    lucide.createIcons();

    // [2] Master Data Management State (สำหรับใช้ Auto-suggest ใน View 4 และจัดการใน View 5)
    let vendorMasterList = [
      { code: '785124', name: 'ABC Company Ltd.' },
      { code: '785324', name: 'ABZ Cooperation' },
      { code: '785882', name: 'Absolute Supply Co., Ltd.' },
      { code: '991002', name: 'Global Logistics Corp' }
    ];

    // [3] View Navigation Routing Logic
    const views = {
      login: document.getElementById('view-login'),
      dashboard: document.getElementById('view-dashboard'),
      createRequest: document.getElementById('view-create-request'),
      vendorMaster: document.getElementById('view-vendor-master')
    };
    const appHeader = document.getElementById('app-header');

    function navigateTo(targetView) {
      Object.values(views).forEach(v => v.classList.add('hidden'));

      if (targetView === 'login') {
        appHeader.classList.add('hidden');
      } else {
        appHeader.classList.remove('hidden');
      }

      if (views[targetView]) {
        views[targetView].classList.remove('hidden');
      }

      if (targetView === 'vendorMaster') {
        renderVendorMasterTable();
      }

      window.scrollTo(0, 0);
    }

    // [4] View 3: Review Modal Logic
    const reviewOverlay = document.getElementById('review-modal-overlay');
    const reviewModal = document.getElementById('review-modal');

    function openReviewModal(code, name, po, isReadOnly = false) {
      document.getElementById('modal-sub-header').textContent = `Supplier Code: ${code} | ${po}`;
      document.getElementById('modal-supplier-name').textContent = name;
      document.getElementById('modal-po-no').textContent = po;
      document.getElementById('rejectReason').value = '';

      const actionsDiv = document.getElementById('modal-actions');
      if (isReadOnly) {
        actionsDiv.classList.add('hidden');
      } else {
        actionsDiv.classList.remove('hidden');
      }

      reviewOverlay.classList.remove('hidden');
      reviewModal.classList.remove('hidden');
      document.body.style.overflow = 'hidden';
    }

    function closeReviewModal() {
      reviewOverlay.classList.add('hidden');
      reviewModal.classList.add('hidden');
      document.body.style.overflow = 'auto';
    }

    function handleDecision(type) {
      if (type === 'approve') {
        alert('อนุมัติเอกสารเรียบร้อยแล้ว');
      } else {
        const reason = document.getElementById('rejectReason').value;
        if (!reason.trim()) {
          alert('กรุณาระบุเหตุผลการปฏิเสธ (Reject Reason)');
          return;
        }
        alert(`ปฏิเสธเอกสารเรียบร้อยแล้ว\nเหตุผล: ${reason}`);
      }
      closeReviewModal();
    }

    // [5] View 4: Auto-Suggestion Logic (Supplier Code & Name)
    const codeInput = document.getElementById('reqSupplierCode');
    const nameInput = document.getElementById('reqSupplierName');
    const codeSuggestBox = document.getElementById('codeSuggestBox');
    const nameSuggestBox = document.getElementById('nameSuggestBox');

    // Code Auto-suggest
    codeInput.addEventListener('input', (e) => {
      const val = e.target.value.trim().toLowerCase();
      codeSuggestBox.innerHTML = '';
      
      if (!val) {
        codeSuggestBox.classList.add('hidden');
        return;
      }

      const matches = vendorMasterList.filter(item => item.code.toLowerCase().includes(val));
      if (matches.length > 0) {
        matches.forEach(item => {
          const div = document.createElement('div');
          div.className = 'p-2.5 hover:bg-slate-50 cursor-pointer text-xs flex justify-between items-center';
          div.innerHTML = `<span class="font-bold text-blue-600">${item.code}</span><span class="text-slate-500">${item.name}</span>`;
          div.onclick = () => {
            codeInput.value = item.code;
            nameInput.value = item.name;
            codeSuggestBox.classList.add('hidden');
          };
          codeSuggestBox.appendChild(div);
        });
        codeSuggestBox.classList.remove('hidden');
      } else {
        codeSuggestBox.classList.add('hidden');
      }
    });

    // Name Auto-suggest
    nameInput.addEventListener('input', (e) => {
      const val = e.target.value.trim().toLowerCase();
      nameSuggestBox.innerHTML = '';

      if (!val) {
        nameSuggestBox.classList.add('hidden');
        return;
      }

      const matches = vendorMasterList.filter(item => item.name.toLowerCase().includes(val));
      if (matches.length > 0) {
        matches.forEach(item => {
          const div = document.createElement('div');
          div.className = 'p-2.5 hover:bg-slate-50 cursor-pointer text-xs flex justify-between items-center';
          div.innerHTML = `<span class="font-bold text-slate-800">${item.name}</span><span class="text-blue-600 font-semibold">${item.code}</span>`;
          div.onclick = () => {
            nameInput.value = item.name;
            codeInput.value = item.code;
            nameSuggestBox.classList.add('hidden');
          };
          nameSuggestBox.appendChild(div);
        });
        nameSuggestBox.classList.remove('hidden');
      } else {
        nameSuggestBox.classList.add('hidden');
      }
    });

    // Hide suggestion boxes on click outside
    document.addEventListener('click', (e) => {
      if (!codeInput.contains(e.target)) codeSuggestBox.classList.add('hidden');
      if (!nameInput.contains(e.target)) nameSuggestBox.classList.add('hidden');
    });

    // [6] View 5: Vendor Master Data Management Logic
    function renderVendorMasterTable() {
      const tbody = document.getElementById('vendor-master-tbody');
      const countBadge = document.getElementById('vendor-count');
      tbody.innerHTML = '';
      countBadge.textContent = `${vendorMasterList.length} รายการ`;

      vendorMasterList.forEach((v, index) => {
        const tr = document.createElement('tr');
        tr.className = 'hover:bg-slate-50';
        tr.innerHTML = `
          <td class="px-4 py-3 font-bold text-slate-800">${v.code}</td>
          <td class="px-4 py-3 text-slate-600">${v.name}</td>
          <td class="px-4 py-3 text-right">
            <button onclick="deleteVendor(${index})" class="text-red-500 hover:text-red-700 p-1">
              <i data-lucide="trash-2" class="w-4 h-4"></i>
            </button>
          </td>
        `;
        tbody.appendChild(tr);
      });
      lucide.createIcons();
    }

    document.getElementById('addVendorForm').addEventListener('submit', (e) => {
      e.preventDefault();
      const code = document.getElementById('newVendorCode').value.trim();
      const name = document.getElementById('newVendorName').value.trim();

      if (code && name) {
        vendorMasterList.push({ code, name });
        document.getElementById('newVendorCode').value = '';
        document.getElementById('newVendorName').value = '';
        renderVendorMasterTable();
        alert('เพิ่มข้อมูล Vendor Master สำเร็จ');
      }
    });

    function deleteVendor(index) {
      if (confirm('คุณต้องการลบรายการ Master นี้ใช่หรือไม่?')) {
        vendorMasterList.splice(index, 1);
        renderVendorMasterTable();
      }
    }

    // [7] Form Submit Handlers
    document.getElementById('loginForm').addEventListener('submit', (e) => {
      e.preventDefault();
      navigateTo('dashboard');
    });

    document.getElementById('logoutBtn').addEventListener('click', () => {
      navigateTo('login');
    });

    document.getElementById('createRequestForm').addEventListener('submit', (e) => {
      e.preventDefault();
      alert('สร้างคำขอเอกสาร (Request Document) เรียบร้อยแล้ว!');
      navigateTo('dashboard');
    });
  </script>
