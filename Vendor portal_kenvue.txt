  <!-- ==========================================

  <script>
    // [1] Initialize Lucide Icons
    lucide.createIcons();

    // [2] Application Routing Views
    const views = {
      login: document.getElementById('view-login'),
      summary: document.getElementById('view-summary'),
      upload: document.getElementById('view-upload')
    };
    const appHeader = document.getElementById('app-header');

    function navigateTo(targetView) {
      Object.values(views).forEach(view => view.classList.add('hidden'));

      if (targetView === 'login') {
        appHeader.classList.add('hidden');
      } else {
        appHeader.classList.remove('hidden');
      }

      if (views[targetView]) {
        views[targetView].classList.remove('hidden');
      }
      
      window.scrollTo(0, 0);
    }

    // [3] Preview Modal Control Logic
    const modalOverlay = document.getElementById('preview-modal-overlay');
    const modalWindow = document.getElementById('preview-modal');
    const modalPoText = document.getElementById('modal-po-reference');

    function openPreviewModal(poReference) {
      modalPoText.textContent = `Reference: ${poReference}`;
      modalOverlay.classList.remove('hidden');
      modalWindow.classList.remove('hidden');
      document.body.style.overflow = 'hidden';
    }

    function closePreviewModal() {
      modalOverlay.classList.add('hidden');
      modalWindow.classList.add('hidden');
      document.body.style.overflow = 'auto';
    }

    modalOverlay.addEventListener('click', closePreviewModal);

    // [4] View 3 File Selection & Drag/Drop Logic
    const fileInputs = document.querySelectorAll('.file-input');
    const dropzones = document.querySelectorAll('.dropzone');

    function handleFileSelection(inputElement, file) {
      if (!file) return;

      const container = inputElement.closest('.bg-white');
      const fileNameDisplay = container.querySelector('.file-name');
      const statusBadgeId = inputElement.getAttribute('data-status-id');
      const statusBadge = document.getElementById(statusBadgeId);

      fileNameDisplay.textContent = `Attached: ${file.name} (${(file.size / (1024 * 1024)).toFixed(2)} MB)`;
      fileNameDisplay.classList.remove('text-slate-400');
      fileNameDisplay.classList.add('text-emerald-600', 'font-semibold');

      if (statusBadge) {
        statusBadge.textContent = 'Status: Attached';
        statusBadge.className = 'inline-flex items-center px-2.5 py-1 rounded-full text-xs font-medium bg-emerald-100 text-emerald-800 self-start';
      }
    }

    fileInputs.forEach(input => {
      input.addEventListener('change', (e) => {
        const file = e.target.files[0];
        handleFileSelection(input, file);
      });
    });

    dropzones.forEach(zone => {
      const input = zone.querySelector('.file-input');

      ['dragenter', 'dragover'].forEach(eventName => {
        zone.addEventListener(eventName, (e) => {
          e.preventDefault();
          zone.classList.add('border-blue-500', 'bg-blue-50');
        });
      });

      ['dragleave', 'drop'].forEach(eventName => {
        zone.addEventListener(eventName, (e) => {
          e.preventDefault();
          zone.classList.remove('border-blue-500', 'bg-blue-50');
        });
      });

      zone.addEventListener('drop', (e) => {
        const files = e.dataTransfer.files;
